solutionCode: |
  import java.text.Collator;
  import java.util.Locale;
  import java.util.Scanner;

  public class Mod2_Exercise3 {
      public static void main(String[] args) {
          Scanner scanner = new Scanner(System.in);

          System.out.print("Enter language tag (e.g., 'fr-FR' for France, 'de-DE' for Germany): ");
          String langTag = scanner.nextLine();
          Locale userLocale = Locale.forLanguageTag(langTag);

          System.out.print("Enter first string: ");
          String str1 = scanner.nextLine();
          System.out.print("Enter second string: ");
          String str2 = scanner.nextLine();

          Collator localeCollator = Collator.getInstance(userLocale);
          int result = localeCollator.compare(str1, str2);

          System.out.println("\nComparison Results (Locale: " + userLocale.getDisplayName() + "):");
          if (result == 0) {
              System.out.println("'" + str1 + "' is EQUIVALENT to '" + str2 + "'");
          } else if (result < 0) {
              System.out.println("'" + str1 + "' comes BEFORE '" + str2 + "'");
          } else {
              System.out.println("'" + str1 + "' comes AFTER '" + str2 + "'");
          }

          scanner.close();
      }
  }
