  - "Search for author 'harper lee' using a case‑insensitive collation (`utf8mb4_unicode_ci`)."
  - "Search for title 'Les Misérables' using a binary case‑sensitive collation (`utf8mb4_bin`)."
solutionLang: "sql"
solutionCode: |
  -- Sort books by title
  SELECT * FROM books
  ORDER BY title;

  -- Case‑insensitive search for author (using the default collation)
  SELECT * FROM books
  WHERE author COLLATE utf8mb4_unicode_ci = 'harper lee';

  -- Case‑sensitive search for title
  SELECT * FROM books
  WHERE title COLLATE utf8mb4_bin = 'Les Misérables';
