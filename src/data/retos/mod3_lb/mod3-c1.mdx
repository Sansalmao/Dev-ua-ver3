solutionCode: |
  import java.io.BufferedReader;
  import java.io.FileInputStream;
  import java.io.IOException;
  import java.io.InputStreamReader;
  import java.text.Collator;
  import java.text.Normalizer;
  import java.util.ArrayList;
  import java.util.List;
  import java.util.Locale;

  public class Mod3_Exercise1 {
      public static void main(String[] args) throws IOException {
          // Read list of names from UTF-8 file
          List<String> listOfStrings = new ArrayList<>();
          try (BufferedReader bf = new BufferedReader(
                  new InputStreamReader(new FileInputStream("names.txt"), "UTF-8"))) {
              String line;
              while ((line = bf.readLine()) != null) {
                  listOfStrings.add(line);
              }
          }

          String[] names = listOfStrings.toArray(new String[0]);

          // Search for a given name
          String name = "Renee"; // This could come from user input

          String normalizedSearch = Normalizer.normalize(name, Normalizer.Form.NFD);
          Collator collator = Collator.getInstance(Locale.getDefault());
          collator.setStrength(Collator.PRIMARY);

          boolean found = false;
          String match = "";

          for (String nm : names) {
              String normalizedName = Normalizer.normalize(nm, Normalizer.Form.NFD);
              if (collator.compare(normalizedSearch, normalizedName) == 0) {
                  match = nm;
                  found = true;
                  break;
              }
          }

          if (found) {
              System.out.println("Matched with " + match);
          } else {
              System.out.println("Username not found");
          }
      }
  }
