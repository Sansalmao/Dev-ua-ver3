solutionCode: |
  import java.text.Collator;
  import java.text.Normalizer;
  import java.util.ArrayList;
  import java.util.Arrays;
  import java.util.List;
  import java.util.Locale;
  import java.util.Scanner;

  public class Mod3_Exercise2 {
      public static void main(String[] args) {
          String[] names = {
              "José", "Håkon", "Renée", "Zoë", "Tom", "Søren",
              "يوسف", "أحمد", "大谷", "Søren"
          };

          // Remove duplicates using primary-strength collation
          Collator collator = Collator.getInstance(Locale.getDefault());
          collator.setStrength(Collator.PRIMARY);

          List<String> uniqueNames = new ArrayList<>();
          for (String name : names) {
              boolean isDuplicate = false;
              String normName = Normalizer.normalize(name, Normalizer.Form.NFD);
              for (String existing : uniqueNames) {
                  String normExisting = Normalizer.normalize(existing, Normalizer.Form.NFD);
                  if (collator.compare(normName, normExisting) == 0) {
                      isDuplicate = true;
                      break;
                  }
              }
              if (!isDuplicate) {
                  uniqueNames.add(name);
              }
          }

          // Get user's locale for sorting
          System.out.print("Please enter your specified locale (e.g., 'da-DK'): ");
          Scanner scanner = new Scanner(System.in);
          String localeTag = scanner.nextLine();
          scanner.close();

          Locale userLocale = Locale.forLanguageTag(localeTag);
          Collator sortCollator = Collator.getInstance(userLocale);

          String[] sortedNames = uniqueNames.toArray(new String[0]);
          Arrays.sort(sortedNames, sortCollator);

          System.out.println("Sorted names according to " + userLocale.getDisplayName() + ":");
          for (String name : sortedNames) {
              System.out.println("\t• " + name);
          }
      }
  }
