```python
    return root
```

## Unicode Collation Algorithm for Sorting of Unicode Strings

The Unicode Collation Algorithm (UCA) is a set of rules defined by the Unicode Consortium for sorting Unicode strings. Key components:

- **Code Point Comparison**
- **Diacritic Sensitivity**
- **Case Sensitivity**
- **Linguistic Context**
- **Collation Element**
- **Collation Rules**

```python
import unicodedata

def unicode_aware_bubble_sort(strings):
    n = len(strings)
    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            if compare_strings(strings[j], strings[j + 1]) > 0:
                strings[j], strings[j + 1] = strings[j + 1], strings[j]
                swapped = True
        if not swapped:
            break

def compare_strings(str1, str2):
    str1_norm = unicodedata.normalize('NFC', str1)
    str2_norm = unicodedata.normalize('NFC', str2)
    return (str1_norm > str2_norm) - (str1_norm < str2_norm)
```

```java
    public static void unicodeAwareBubbleSort(String[] strings) {
        int n = strings.length;
        for (int i = 0; i < n; i++) {
            boolean swapped = false;
            for (int j = 0; j < n - i - 1; j++) {
                if (compareStrings(strings[j], strings[j + 1]) > 0) {

                    strings[j]   = strings[j + 1];
                    strings[j + 1] = temp;

    }

    public static int compareStrings(String str1, String str2) {
        Collator collator = Collator.getInstance(Locale.getDefault());
        return collator.compare(str1, str2);
    }
```

- Similar to Bubble Sort, use Unicode-aware comparison functions to compare and select the appropriate string based on Unicode code points or collation order.
- Apply Unicode normalization to ensure consistent sorting results.

```python
import unicodedata

def compare_strings(str1, str2):
    str1_norm = unicodedata.normalize('NFD', str1)
    str2_norm = unicodedata.normalize('NFD', str2)
    return ord(str1_norm[0]) - ord(str2_norm[0])

def selection_sort(arr):
    n = len(arr)
    for i in range(n):
        min_idx = i
        for j in range(i + 1, n):
            if compare_strings(arr[j], arr[min_idx]) < 0:
                min_idx = j
        arr[i], arr[min_idx] = arr[min_idx], arr[i]
    return arr

strings = [
    "अनुच्छेद", "Éclair", "Österreich",
    "中国", "Россия", "日本", "ሉሲ"
]
sorted_strings = selection_sort(strings)
for item in sorted_strings:
    print(item)
```

```java
    public static void selectionSort(String[] arr) {
        int n = arr.length;

    }
}
```

<span class="hl">Insertion Sort:</span>

- Utilize Unicode-aware comparison functions to compare and insert strings into their correct positions within the sorted portion of the list.
- Normalize Unicode strings using Unicode normalization to maintain consistency during the insertion process.

```python
import unicodedata

def compare_strings(str1, str2):
    s1 = unicodedata.normalize('NFD', str1)
    s2 = unicodedata.normalize('NFD', str2)
    return ord(s1[0]) - ord(s2[0])

def insertion_sort(arr):
    n = len(arr)
    for i in range(1, n):
        key = arr[i]
        j = i - 1
        while j >= 0 and compare_strings(arr[j], key) > 0:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key
    return arr
```

```java
import java.text.Normalizer;
import java.util.Comparator;

public class StringComparator {
    public static String[] insertionSort(String[] arr) {
        int n = arr.length;
        for (int i = 1; i < n; i++) {
            String key = arr[i];
            int j = i - 1;
            while (j >= 0 && compareStrings(arr[j], key) > 0) {
                arr[j + 1] = arr[j]; j--;
            }
            arr[j + 1] = key;
        }
        return arr;
    }

    public static int compareStrings(String str1, String str2) {
        String s1 = Normalizer.normalize(str1, Normalizer.Form.NFD);
        String s2 = Normalizer.normalize(str2, Normalizer.Form.NFD);
        return Character.codePointAt(s1, 0) - Character.codePointAt(s2, 0);
    }
}
```

```python
    if left  < n and compare_strings(arr[left],  arr[largest]) > 0: largest = left
```

```java
        if (left  < n && compareStrings(arr[left],  arr[largest]) > 0) largest = left;
        if (right < n && compareStrings(arr[right], arr[largest]) > 0) largest = right;
        if (largest != i) {
            String temp = arr[i]; arr[i] = arr[largest]; arr[largest] = temp;
            heapify(arr, n, largest);
        }
    }

        int n = arr.length;
        for (int i = n/2 - 1; i >= 0; i--) heapify(arr, n, i);
        for (int i = n - 1; i > 0; i--) {
            String temp = arr[0]; arr[0] = arr[i]; arr[i] = temp;
            heapify(arr, i, 0);
        }
        return arr;
```

- Is a hybrid sorting algorithm that works well with various types of data, including Unicode strings.
- Employ Unicode-aware comparison functions or collation keys during the sorting process to ensure accurate sorting of Unicode strings.
- Normalize Unicode strings for consistent sorting results.

```python
    right_arr = arr[mid + 1:right + 1]

        else:

        k += 1

    while j < len(right_arr): arr[k] = right_arr[j]; j += 1; k += 1

def tim_sort(arr):
    n = len(arr); min_run = 32
    for i in range(0, n, min_run):
        insertion_sort(arr, i, min(i + min_run - 1, n - 1))
    size = min_run
    while size < n:
        for start in range(0, n, size * 2):
            mid = min(start + size - 1, n - 1)
            end = min(start + size * 2 - 1, n - 1)
            merge(arr, start, mid, end)
        size *= 2
    return arr
```

```java
        String s1 = Normalizer.normalize(str1, Normalizer.Form.NFD);

    public static void insertionSort(String[] arr, int left, int right) {
        for (int i = left + 1; i <= right; i++) {
            String key = arr[i]; int j = i - 1;
            while (j >= left && compareStrings(arr[j], key) > 0) {
                arr[j + 1] = arr[j]; j--;
            }
            arr[j + 1] = key;
```

## Searching Through Unicode Text (Normalization + Collation)
